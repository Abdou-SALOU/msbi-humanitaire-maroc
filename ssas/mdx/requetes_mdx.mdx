-- =====================================================================
-- MDX 1 - Personnes accueillies au Maroc par annee et statut (2019-2025)
-- En 2025 : 15 033 demandeurs d'asile ; 7 304 refugies ; total 22 337.
-- =====================================================================
SELECT
    NON EMPTY
        [Groupe Population].[Groupe Key].MEMBERS ON COLUMNS,
    {
        [Date].[Annee].&[2019] :
        [Date].[Annee].&[2025]
    } ON ROWS
FROM [Humanitaire Maroc]
WHERE (
    [Measures].[Personnes],
    FILTER(
        [Pays Asile].[Pays Key].MEMBERS,
        [Pays Asile].[Pays Key].CURRENTMEMBER.NAME = "Morocco"
    ).ITEM(0)
);


-- =====================================================================
-- MDX 2 - Top 5 des pays d'origine des personnes accueillies au Maroc
-- 2025 : Sudan 5291, Syria 5216, Guinea 2366, Cote d'Ivoire 1780,
--        Senegal 1584.
-- =====================================================================
SELECT
    [Measures].[Personnes] ON COLUMNS,
    TOPCOUNT(
        [Pays Origine].[Pays Key].LEVELS(1).MEMBERS,
        5,
        [Measures].[Personnes]
    ) ON ROWS
FROM [Humanitaire Maroc]
WHERE (
    [Date].[Annee].&[2025],
    FILTER(
        [Pays Asile].[Pays Key].MEMBERS,
        [Pays Asile].[Pays Key].CURRENTMEMBER.NAME = "Morocco"
    ).ITEM(0)
);


-- =====================================================================
-- MDX 3 - Top 5 des pays d'accueil des personnes originaires du Maroc
-- 2025 : Italy 13188, Spain 3790, France 2676, Germany 2606,
--        United States of America 1824.
-- =====================================================================
SELECT
    [Measures].[Personnes] ON COLUMNS,
    TOPCOUNT(
        [Pays Asile].[Pays Key].LEVELS(1).MEMBERS,
        5,
        [Measures].[Personnes]
    ) ON ROWS
FROM [Humanitaire Maroc]
WHERE (
    [Date].[Annee].&[2025],
    FILTER(
        [Pays Origine].[Pays Key].MEMBERS,
        [Pays Origine].[Pays Key].CURRENTMEMBER.NAME = "Morocco"
    ).ITEM(0)
);


-- =====================================================================
-- MDX 4 - Pluviometrie par region en 2025
-- Anomalie = 100 pour la normale ; ordre croissant de l'anomalie.
-- Marrakech-Safi : 89,4 ; Rabat-Sale-Kenitra : 116,7.
-- =====================================================================
SELECT
    {
        [Measures].[Pluie moyenne],
        [Measures].[Anomalie pluie]
    } ON COLUMNS,
    NON EMPTY
        ORDER(
            [Region].[Region Key].LEVELS(1).MEMBERS,
            [Measures].[Anomalie pluie],
            BASC
        ) ON ROWS
FROM [Humanitaire Maroc]
WHERE [Date].[Annee].&[2025];


-- =====================================================================
-- MDX 5 - Taux de chomage 2024 par region et milieu
-- Le membre All = total recalcule pour les mesures de ratio.
-- Total national : 21,3 ; Guelmim-Oued Noun rural : 38,2.
-- =====================================================================
SELECT
    NON EMPTY
        [Milieu].[Milieu Key].MEMBERS ON COLUMNS,
    NON EMPTY
        [Region].[Region Key].MEMBERS ON ROWS
FROM [Humanitaire Maroc]
WHERE (
    [Measures].[Taux de chômage],
    [Date].[Annee].&[2024]
);


-- =====================================================================
-- MDX 6 - Pauvrete multidimensionnelle par region : 2014 vs 2024
-- WITH MEMBER : mesures temporaires propres a cette requete.
-- Marrakech-Safi : 15,8 -> 7,9 ; evolution -7,9 points.
-- =====================================================================
WITH
    MEMBER [Measures].[Pauvreté 2014] AS
        (
            [Measures].[Taux de pauvreté],
            [Date].[Annee].&[2014]
        ),
        FORMAT_STRING = "0.0"

    MEMBER [Measures].[Pauvreté 2024] AS
        (
            [Measures].[Taux de pauvreté],
            [Date].[Annee].&[2024]
        ),
        FORMAT_STRING = "0.0"

    MEMBER [Measures].[Évolution (points)] AS
        [Measures].[Pauvreté 2024]
        - [Measures].[Pauvreté 2014],
        FORMAT_STRING = "+0.0;-0.0"

SELECT
    {
        [Measures].[Pauvreté 2014],
        [Measures].[Pauvreté 2024],
        [Measures].[Évolution (points)]
    } ON COLUMNS,
    ORDER(
        [Region].[Region Key].LEVELS(1).MEMBERS,
        [Measures].[Évolution (points)],
        BASC
    ) ON ROWS
FROM [Humanitaire Maroc]
WHERE [Milieu].[Milieu Key].&[1];


-- =====================================================================
-- MDX 7 - Evenements de conflit par annee et type (2020-2025)
-- Manifestations : 3654 en 2024 et 2776 en 2025.
-- =====================================================================
SELECT
    [Type Evenement].[Type Key].LEVELS(1).MEMBERS ON COLUMNS,
    {
        [Date].[Annee].&[2020] :
        [Date].[Annee].&[2025]
    } ON ROWS
FROM [Humanitaire Maroc]
WHERE [Measures].[Événements];


-- =====================================================================
-- MDX 8 - Evolution annuelle des personnes accueillies au Maroc
-- PrevMember est utilise dans la mesure calculee du cube.
-- 2020 : +39,0 ; 2022 : -0,9 ; 2024 : -4,0 ; 2025 : +18,5.
-- =====================================================================
SELECT
    {
        [Measures].[Personnes],
        [Measures].[Évolution annuelle personnes]
    } ON COLUMNS,
    {
        [Date].[Annee].&[2019] :
        [Date].[Annee].&[2025]
    } ON ROWS
FROM [Humanitaire Maroc]
WHERE
    FILTER(
        [Pays Asile].[Pays Key].MEMBERS,
        [Pays Asile].[Pays Key].CURRENTMEMBER.NAME = "Morocco"
    ).ITEM(0);
