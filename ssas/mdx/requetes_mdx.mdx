-- MDX 1 - Personnes accueillies au Maroc, par année et par statut
-- Notions : axes COLUMNS/ROWS, plage de membres (:), découpage WHERE (slicer), NON EMPTY
-- Attendu : 2025 -> 15 033 demandeurs d'asile, 7 304 réfugiés
SELECT
    NON EMPTY [Groupe Population].[Groupe].[Groupe].MEMBERS ON COLUMNS,
    { [Date].[Annee].&[2019] : [Date].[Annee].&[2025] } ON ROWS
FROM [Humanitaire Maroc]
WHERE ( [Measures].[Personnes], [Pays Asile].[Pays].[Morocco] );


-- MDX 2 - Top 5 des pays d'origine des personnes accueillies au Maroc en 2025
-- Notions : TOPCOUNT, dimension à deux rôles (Pays Origine / Pays Asile)
-- Attendu : Sudan 5 291, Syria 5 216, Guinea 2 366, Côte d'Ivoire 1 780, Senegal 1 584
SELECT
    [Measures].[Personnes] ON COLUMNS,
    TOPCOUNT([Pays Origine].[Pays].[Pays].MEMBERS, 5, [Measures].[Personnes]) ON ROWS
FROM [Humanitaire Maroc]
WHERE ( [Date].[Annee].&[2025], [Pays Asile].[Pays].[Morocco] );


-- MDX 3 - Top 5 des pays où des Marocains ont demandé l'asile ou sont réfugiés (2025)
-- Même dimension, rôle inversé : le Maroc devient le pays d'ORIGINE
-- Attendu : Italy 13 188, Spain 3 790, France 2 676, Germany 2 606, United States 1 824
SELECT
    [Measures].[Personnes] ON COLUMNS,
    TOPCOUNT([Pays Asile].[Pays].[Pays].MEMBERS, 5, [Measures].[Personnes]) ON ROWS
FROM [Humanitaire Maroc]
WHERE ( [Date].[Annee].&[2025], [Pays Origine].[Pays].[Morocco] );


-- MDX 4 - Pluie de 2025 par région : moyenne par décade et anomalie (100 = normale)
-- Notions : membres calculés du cube (moyenne pondérée), ORDER, NON EMPTY
-- Attendu : Marrakech-Safi la plus sèche (89,4), Rabat-Salé-Kénitra la plus arrosée (116,7)
SELECT
    { [Measures].[Pluie moyenne], [Measures].[Anomalie pluie] } ON COLUMNS,
    NON EMPTY ORDER([Region].[Region].[Region].MEMBERS, [Measures].[Anomalie pluie], BASC) ON ROWS
FROM [Humanitaire Maroc]
WHERE [Date].[Annee].&[2025];


-- MDX 5 - Taux de chômage 2024 par région, ensemble / urbain / rural
-- Notions : le membre [All] = total recalculé (le taux national n'est stocké nulle part)
-- Attendu : Maroc (All, All) = 21,3 % ; Guelmim-Oued Noun rural = 38,2 %
SELECT
    { [Milieu].[Milieu].[All], [Milieu].[Milieu].[Urbain], [Milieu].[Milieu].[Rural] } ON COLUMNS,
    { [Region].[Region].[All], [Region].[Region].[Region].MEMBERS } ON ROWS
FROM [Humanitaire Maroc]
WHERE [Measures].[Taux de chômage];


-- MDX 6 - Pauvreté multidimensionnelle : 2014, 2024 et évolution par région
-- Notions : WITH MEMBER (membres calculés propres à la requête), tuples
-- Attendu : baisse partout ; Marrakech-Safi -7,9 points (15,8 -> 7,9)
WITH
    MEMBER [Measures].[Pauvreté 2014] AS ([Measures].[Taux de pauvreté], [Date].[Annee].&[2014]), FORMAT_STRING = "0.0"
    MEMBER [Measures].[Pauvreté 2024] AS ([Measures].[Taux de pauvreté], [Date].[Annee].&[2024]), FORMAT_STRING = "0.0"
    MEMBER [Measures].[Évolution (points)] AS [Measures].[Pauvreté 2024] - [Measures].[Pauvreté 2014], FORMAT_STRING = "+0.0;-0.0"
SELECT
    { [Measures].[Pauvreté 2014], [Measures].[Pauvreté 2024], [Measures].[Évolution (points)] } ON COLUMNS,
    ORDER([Region].[Region].[Region].MEMBERS, [Measures].[Évolution (points)], BASC) ON ROWS
FROM [Humanitaire Maroc]
WHERE [Milieu].[Milieu].[Ensemble];


-- MDX 7 - Événements de conflit par année et par type (2020-2025)
-- Attendu : 3 654 manifestations en 2024 (record), 2 776 en 2025
SELECT
    [Type Evenement].[Type].[Type].MEMBERS ON COLUMNS,
    { [Date].[Annee].&[2020] : [Date].[Annee].&[2025] } ON ROWS
FROM [Humanitaire Maroc]
WHERE [Measures].[Événements];


-- MDX 8 - Évolution annuelle des personnes accueillies au Maroc
-- Notions : PrevMember (navigation dans le temps), membre calculé du cube
-- Attendu : 2019 = 9 721 ; 2020 = 13 516 (+39,0 %) ; 2022 : -0,9 % ; 2024 : -4,0 % ; 2025 = 22 337 (+18,5 %)
SELECT
    { [Measures].[Personnes], [Measures].[Évolution annuelle personnes] } ON COLUMNS,
    { [Date].[Annee].&[2019] : [Date].[Annee].&[2025] } ON ROWS
FROM [Humanitaire Maroc]
WHERE [Pays Asile].[Pays].[Morocco];
